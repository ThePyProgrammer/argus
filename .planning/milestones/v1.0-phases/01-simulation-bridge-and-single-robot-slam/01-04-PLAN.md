       ```python
       import rerun as rr
       import numpy as np

       class RerunVisualizer:
           def __init__(self, app_name: str = "slam_viz"):
               rr.init(app_name, spawn=True)

           def log_frame(self, frame):
               """Log RGB and depth images from a SensorFrame."""
               rr.log("camera/rgb", rr.Image(frame.rgb))
               if frame.depth is not None:
                   rr.log("camera/depth", rr.DepthImage(frame.depth))

           def log_point_cloud(self, points: np.ndarray, colors: np.ndarray | None = None):
               """Log accumulated global point cloud."""
               if len(points) == 0:
                   return
               if colors is not None and len(colors) == len(points):
                   rr.log("map/point_cloud", rr.Points3D(points, colors=colors))
               else:
                   rr.log("map/point_cloud", rr.Points3D(points))

           def log_occupancy_grid(self, voxel_centers: np.ndarray, resolution: float):
               """Log occupied voxels from OctoMap."""
               if len(voxel_centers) == 0:
                   return
               rr.log("map/occupancy", rr.Points3D(
                   voxel_centers,
                   radii=resolution / 2.0,
                   colors=np.tile([0, 200, 0], (len(voxel_centers), 1)),
               ))

           def log_trajectory(self, poses: list[np.ndarray], entity: str = "robot/slam_trajectory"):
               """Log robot trajectory as line strip."""
               if len(poses) < 2:
                   return
               positions = np.array([p[:3, 3] for p in poses])
               rr.log(entity, rr.LineStrips3D([positions]))

           def log_robot_pose(self, pose: np.ndarray, entity: str = "robot/current"):
               """Log current robot transform."""
               rr.log(entity, rr.Transform3D(
                   translation=pose[:3, 3],
                   mat3x3=pose[:3, :3],
               ))
       ```

    3. Create `src/main.py` -- the entry point that wires everything together:
       ```python
       #!/usr/bin/env python3
       """Main entry point: SimWorld bridge -> SLAM -> OctoMap -> Rerun visualization."""
       import argparse
       import sys
       import time
       import numpy as np

       from src.bridge.sim_bridge import SimWorldGymBridge
       from src.bridge.env_config import SimWorldEnvConfig
       from src.bridge.sensor_types import CameraIntrinsics
       from src.slam.slam_pipeline import SLAMPipeline
       from src.slam.octomap_builder import OctoMapBuilder
       from src.metrics.drift_metrics import compute_drift_metrics
       from src.metrics.ground_truth import GroundTruthCollector
       from src.viz.rerun_viz import RerunVisualizer

       def parse_args():
           parser = argparse.ArgumentParser(description="Single-robot SLAM in SimWorld")
           parser.add_argument("--control", choices=["teleop", "waypoint", "random"],
                               default="teleop", help="Control mode")
           parser.add_argument("--max-steps", type=int, default=1000,
                               help="Maximum simulation steps")
           parser.add_argument("--octomap-resolution", type=float, default=0.1,
                               help="OctoMap voxel resolution in meters")
           parser.add_argument("--octomap-interval", type=int, default=5,
                               help="Insert into OctoMap every N frames")
           parser.add_argument("--no-viz", action="store_true",
                               help="Disable Rerun visualization")
           return parser.parse_args()

       def create_controller(mode: str):
           if mode == "teleop":
               from src.control.teleop import TeleopController
               return TeleopController()
           elif mode == "waypoint":
               from src.control.waypoint_runner import WaypointRunner
               return WaypointRunner(waypoints=[
                   np.array([5.0, 0.0, 0.0]),
                   np.array([5.0, 5.0, 0.0]),
                   np.array([0.0, 5.0, 0.0]),
                   np.array([0.0, 0.0, 0.0]),
               ])
           elif mode == "random":
               from src.control.random_walk import RandomWalkController
               return RandomWalkController(seed=42)
           else:
               raise ValueError(f"Unknown control mode: {mode}")

       def main():
           args = parse_args()

           # Initialize components
           config = SimWorldEnvConfig()
           bridge = SimWorldGymBridge(config)

           # Camera intrinsics -- UPDATE after SimWorld discovery
           # These are placeholder values; replace with actual intrinsics from discovery
           intrinsics = CameraIntrinsics(fx=320.0, fy=320.0, cx=320.0, cy=240.0,
                                          width=640, height=480)

           slam = SLAMPipeline(intrinsics)
           octomap = OctoMapBuilder(resolution=args.octomap_resolution)
           gt_collector = GroundTruthCollector()
           viz = RerunVisualizer() if not args.no_viz else None

           controller = create_controller(args.control)
           if hasattr(controller, 'start'):
               controller.start()

           print(f"Starting SLAM with {args.control} control...")
           print(f"Max steps: {args.max_steps}")
           print("Press Ctrl+C to stop\n")

           try:
               # Start simulation
               frame = bridge.start()
               gt_collector.record(frame)

               for step_i in range(args.max_steps):
                   # Get velocity from controller
                   if args.control == "teleop":
                       linear, angular = controller.get_velocity()
                   elif args.control == "waypoint":
                       current_pose = slam.slam_poses[-1] if slam.slam_poses else np.eye(4)
                       linear, angular = controller.get_velocity(current_pose)
                       if controller.is_complete:
                           print("All waypoints reached.")
                           break
                   elif args.control == "random":
                       linear, angular = controller.get_velocity(frame.sim_time)

                   # Step simulation
                   bridge.set_velocity(linear, angular)
                   frame = bridge.step()
                   gt_collector.record(frame)

                   # Process through SLAM
                   slam_pose = slam.process_frame(frame)

                   # Update OctoMap periodically
                   if step_i % args.octomap_interval == 0 and slam.num_frames_processed > 0:
                       cloud_points = slam.get_cloud_points()
                       if len(cloud_points) > 0:
                           sensor_origin = slam_pose[:3, 3]
                           octomap.insert_scan(cloud_points, sensor_origin)

                   # Visualize
                   if viz is not None:
                       viz.log_frame(frame)
                       viz.log_robot_pose(slam_pose)
                       if step_i % 10 == 0:  # Reduce viz frequency for performance
                           cloud_pts = slam.get_cloud_points()
                           cloud_colors = slam.get_cloud_colors()
                           if len(cloud_pts) > 0:
                               viz.log_point_cloud(cloud_pts, cloud_colors)
                           viz.log_trajectory(slam.slam_poses)
                           voxels = octomap.get_occupied_voxels()
                           if len(voxels) > 0:
                               viz.log_occupancy_grid(voxels, octomap.resolution)

                   # Print progress
                   if step_i % 50 == 0:
                       print(f"Step {step_i}: cloud={len(slam.get_cloud_points())} pts, "
                             f"voxels={octomap.num_occupied}, "
                             f"pose=[{slam_pose[0,3]:.2f}, {slam_pose[1,3]:.2f}, {slam_pose[2,3]:.2f}]")

           except KeyboardInterrupt:
               print("\nStopping...")
           finally:
               # Compute and print drift metrics
               if len(gt_collector) >= 2 and len(slam.slam_poses) >= 2:
                   print("\n=== Drift Metrics ===")
                   try:
                       metrics = compute_drift_metrics(
                           slam_poses=slam.slam_poses,
                           gt_poses=gt_collector.poses,
                           timestamps=gt_collector.timestamps,
                       )
                       print(f"ATE RMSE: {metrics['ate_rmse']:.4f} m")
                       print(f"ATE Mean: {metrics['ate_mean']:.4f} m")
                       print(f"RPE RMSE: {metrics['rpe_rmse']:.4f} m")
                       print(f"RPE Mean: {metrics['rpe_mean']:.4f} m")
                   except Exception as e:
                       print(f"Could not compute drift metrics: {e}")

               print(f"\nFinal stats: {slam.num_frames_processed} frames, "
                     f"{len(slam.get_cloud_points())} cloud points, "
                     f"{octomap.num_occupied} occupied voxels")

               # Cleanup
               if hasattr(controller, 'stop'):
                   controller.stop()
               bridge.stop()
               print("Done.")

       if __name__ == "__main__":
           main()
       ```
