    **Add StuckRecovery class** (can be in exploration_loop.py or a small separate file -- keep it in exploration_loop.py for simplicity):

    ```python
    class StuckRecovery:
        """Turn-in-place recovery when robot is physically stuck."""

        def __init__(self, turn_angle: float = np.pi / 2, angular_speed: float = 1.0):
            self._turn_angle = turn_angle
            self._angular_speed = angular_speed
            self._remaining = 0.0
            self._active = False

        def trigger(self):
            """Start a turn recovery. Randomizes direction (left/right)."""
            direction = 1.0 if np.random.random() > 0.5 else -1.0
            self._remaining = self._turn_angle * direction
            self._active = True

        def step(self, dt: float) -> tuple[float, float, float] | None:
            """Get recovery velocity command (vx, vy, omega), or None if not active.

            Returns None when recovery is complete.
            """
            if not self._active:
                return None

            turn = np.sign(self._remaining) * self._angular_speed
            self._remaining -= turn * dt

            if abs(self._remaining) < 0.1:
                self._active = False
                return None

            return (0.0, 0.0, turn)  # zero linear, only angular

        @property
        def is_active(self) -> bool:
            return self._active
    ```
