```cmake
cmake_minimum_required(VERSION 3.14)
project(dso_harness LANGUAGES CXX)

# DSO -- built from source, typically in extern/dso/build

find_path(DSO_INCLUDE dso PATHS ${CMAKE_SOURCE_DIR}/../dso/src REQUIRED)

add_executable(dso_harness main.cpp)
target_include_directories(dso_harness PRIVATE ${DSO_INCLUDE})
target_link_libraries(dso_harness
    PRIVATE
    ${DSO_LIB}
    Eigen3::Eigen
    cppzmq
    msgpack-cxx
)
```

2. Create extern/dso_harness/main.cpp:
   - Parse CLI args: `--config <path>` `--zmq <ipc_endpoint>`
   - Connect ZMQ PAIR socket
   - Initialize DSO's FullSystem
   - Main loop: receive multipart, unpack header, create DSO ImageAndExposure from RGB, call fullSystem->addActiveFrame(), get pose from fullSystem->getAllKFPoses(), pack and send response
   - DSO returns poses in camera-optical frame; harness sends raw pose back to Python (coordinate transform done Python-side)

Key DSO API (from research):
```cpp
#include "FullSystem/FullSystem.h"
#include "IOWrapper/ImageDisplay.h"
#include "util/NumType.h"

auto system = new dso::FullSystem();
system->setGammaFunction(nullptr);  // no photometric calibration

// Per frame:
auto img = new dso::ImageAndExposure(width, height, timestamp);
memcpy(img->image, grayscale_data, width * height * sizeof(float));
system->addActiveFrame(img, frame_id);

// Get poses:
auto poses = system->getAllKFPoses();  // vector<SE3>
// Latest: poses.back().matrix()  // 4x4 Eigen::Matrix4d
```
