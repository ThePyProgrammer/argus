```python
"""Subprocess bridge for out-of-process detector backends (Phase 5 BoxeR skeleton).

Structurally cloned from src/slam/backends/subprocess_bridge.py but with an
entirely separate class per CONTEXT.md D-16:
  - own zmq.Context per instance (not shared with SLAM)
  - endpoint prefix ipc:///tmp/detector_bridge_<pid>_<id> (distinct from slam_bridge_*)
  - hardened msgpack ingress: raw=False, strict_map_key=True
  - Phase 2 reply schema: {ts, inference_ms, n_det, classes, scores, bboxes}

Phase 2 scope: skeleton only — no backend registers this bridge. The handshake
test (Plan 07) exercises spawn/send/recv/kill/cleanup via scripts/echo_detector_worker.py.
Phase 5 (DET-MODELS-03) composes this class into a BoxeR backend.
"""

import logging
import os
import subprocess
from typing import Any

import msgpack
import numpy as np
import zmq

logger = logging.getLogger(__name__)


class SubprocessDetectorBridge:
    HANG_TIMEOUT_MS = 5000

    def __init__(
        self,
        binary_path: str,
        args: list[str] | None = None,
        ipc_endpoint: str | None = None,
        hang_timeout_ms: int | None = None,
    ):
        self._binary = binary_path
        self._args = args or []
        self._endpoint = ipc_endpoint or f"ipc:///tmp/detector_bridge_{os.getpid()}_{id(self)}"
        self._timeout = hang_timeout_ms or self.HANG_TIMEOUT_MS
        self._process: subprocess.Popen | None = None
        self._ctx: zmq.Context | None = None
        self._socket: zmq.Socket | None = None
        self._alive = False

    @property
    def alive(self) -> bool:
        return self._alive

    @property
    def endpoint(self) -> str:
        return self._endpoint

    def start(self) -> None:
        self._ctx = zmq.Context()
        self._socket = self._ctx.socket(zmq.PAIR)
        self._socket.setsockopt(zmq.RCVTIMEO, self._timeout)
        self._socket.setsockopt(zmq.LINGER, 0)
        self._socket.bind(self._endpoint)
        cmd = [self._binary, "--zmq", self._endpoint] + self._args
        logger.info("Spawning detector subprocess: %s", " ".join(cmd))
        self._process = subprocess.Popen(
            cmd, stdout=subprocess.PIPE, stderr=subprocess.PIPE,
        )
        self._alive = True

    def send_frame(
        self,
        rgb: np.ndarray,
        depth: np.ndarray | None,
        timestamp: float,
        params: dict | None = None,
    ) -> dict | None:
        if not self._alive:
            return None
        if self._process and self._process.poll() is not None:
            logger.warning("Detector subprocess exited with code %d", self._process.returncode)
            self._cleanup()
            return None
        try:
            header_dict: dict[str, Any] = {
                "ts": float(timestamp),
                "rgb_shape": list(rgb.shape),
                "rgb_dtype": str(rgb.dtype),
                "depth_shape": None if depth is None else list(depth.shape),
                "depth_dtype": None if depth is None else str(depth.dtype),
                "params": params or {},
            }
            header = msgpack.packb(header_dict)
            parts = [header, rgb.tobytes()]
            if depth is not None:
                parts.append(depth.tobytes())
            self._socket.send_multipart(parts)
            reply = self._socket.recv()
            # Hardened: strict_map_key + raw=False rejects non-string keys and returns str values
            return msgpack.unpackb(reply, raw=False, strict_map_key=True)
        except zmq.Again:
            logger.error("Detector subprocess hung (no response in %dms)", self._timeout)
            self._kill_process()
            return None
        except zmq.ZMQError as exc:
            logger.error("Detector ZMQ error: %s", exc)
            self._kill_process()
            return None
        except Exception as exc:
            logger.error("Unexpected error in detector send_frame: %s", exc)
            self._kill_process()
            return None

    def shutdown(self) -> None:
        self._kill_process()

    def _kill_process(self) -> None:
        self._alive = False
        if self._process:
            try:
                self._process.kill()
                self._process.wait(timeout=2)
            except Exception:
                pass
            self._process = None
        self._cleanup()

    def _cleanup(self) -> None:
        self._alive = False
        if self._socket is not None:
            try:
                self._socket.close()
            except Exception:
                pass
            self._socket = None
        if self._ctx is not None:
            try:
                self._ctx.term()
            except Exception:
                pass
            self._ctx = None
        if self._endpoint.startswith("ipc://"):
            sock_path = self._endpoint.replace("ipc://", "")
            if os.path.exists(sock_path):
                try:
                    os.unlink(sock_path)
                except OSError:
                    pass
```

4. `depth` is optional (RGB-only backends like OWLv2 will skip it).

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| subprocess worker → bridge (msgpack reply ingress) | Worker process may send malformed msgpack either through bug or corruption. |
| bridge → /tmp/ socket file | Localhost file system namespace; other processes on host could race or squat. |
| bridge → subprocess kill signal | Process lifecycle must handle Popen timeout/crash without leaking fds or socket files. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-12 | Tampering | msgpack.unpackb on bridge ingress | mitigate | strict_map_key=True + raw=False reject non-string keys and auto-decode bytes; broad Exception handler falls through to _kill_process. |
| T-02-13 | Tampering | /tmp/detector_bridge_<pid>_<id(self)> socket path | mitigate | pid+id(self) uniqueness; another user's process cannot pre-create the path due to pid uniqueness. See RESEARCH §Security Domain. |
| T-02-14 | Denial of Service | Hung subprocess worker | mitigate | RCVTIMEO=5000 ms + zmq.Again path → _kill_process → subprocess.kill() + cleanup. Bridge never blocks indefinitely. |
| T-02-15 | Resource exhaustion | Leaked /tmp IPC socket files | mitigate | os.unlink in _cleanup; LINGER=0 prevents ctx.term from hanging. Pitfall 3 defended. |
</threat_model>
