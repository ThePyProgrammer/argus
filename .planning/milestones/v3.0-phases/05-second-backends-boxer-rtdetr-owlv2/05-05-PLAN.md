    ```python
    @classmethod
    def set_available(cls, name: str, available: bool, reason: str | None = None) -> None:
        """Override availability for a registered backend (session-scoped, D-04).

        Used by DetectorWorkerPool.on_backend_crash() to lock out a crashed
        backend until the process restarts. Not intended for general use.

        Raises ValueError if ``name`` is not registered.
        """
        if name not in cls._backends:
            raise ValueError(
                f"Unknown detector backend '{name}'. "
                f"Available: {list(cls._backends.keys())}"
            )
        cls._backends[name]["_override_available"] = (bool(available), reason)
    ```
