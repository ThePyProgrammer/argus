```python
class DetectorWorker:
    def __init__(self, robot_id, detector, lifter, intrinsics):
        self._rid = robot_id
        self._detector = detector
        self._lifter = lifter
        self._intrinsics = intrinsics
        self._pending: tuple | None = None
        self._latest: Detections3D | None = None
        self._drops: int = 0
        self._last_submit_sim_time: float = 0.0
        self._lock = threading.Lock()
        self._stop = threading.Event()
        self._thread: threading.Thread | None = None

    def start(self) -> None: ...
    def submit(self, frame, pose, slam_cloud) -> None: ...  # D-04 newest-wins; D-12/D-13 snapshots
    def latest(self) -> Detections3D | None: ...
    def inspect(self) -> dict: ...
    def warmup(self, dummy_frame) -> None: ...
    def reset(self) -> None: ...
    def shutdown(self) -> None: ...
    def _loop(self) -> None: ...
```

```python
def test_defensive_pose_copy():
    from src.bridge.sensor_types import SensorFrame, CameraIntrinsics
    intr = CameraIntrinsics.from_fov(640, 480, 70.0)
    w = DetectorWorker("r0", SlowDetector(), DummyLifter(), intr)

    rgb = np.zeros((480, 640, 3), dtype=np.uint8)
```

| coordinator (submit thread) ↔ worker (_loop thread) | Two threads share _pending and _latest via threading.Lock. |
