```python
#!/usr/bin/env python3
"""Standalone ZMQ PAIR echo worker for SubprocessDetectorBridge handshake test.

Permanent helper per CONTEXT.md D-15 — reusable as a dev harness for Phase 5
BoxeR bring-up. NOT a test-only fixture. Checked into scripts/.

Usage:
    python scripts/echo_detector_worker.py --zmq ipc:///tmp/detector_bridge_12345_67890
    python scripts/echo_detector_worker.py --zmq ipc:///tmp/x --sleep-ms 100

Reply schema matches what SubprocessDetectorBridge.send_frame unpacks:
    {"ts": float, "inference_ms": float, "n_det": 0,
     "classes": [], "scores": [], "bboxes": []}
"""

import argparse
import sys
import time

import msgpack
import zmq


def main() -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("--zmq", required=True, help="ZMQ endpoint (e.g., ipc:///tmp/detector_bridge_...)")
    ap.add_argument("--sleep-ms", type=int, default=0, help="Simulate slow backend (ms per frame)")
    args = ap.parse_args()

    ctx = zmq.Context()
    sock = ctx.socket(zmq.PAIR)
    sock.connect(args.zmq)
    try:
        while True:
            parts = sock.recv_multipart()
            if not parts:
                continue
            header = msgpack.unpackb(parts[0], raw=False)
            if args.sleep_ms:
                time.sleep(args.sleep_ms / 1000.0)
            reply = msgpack.packb({
                "ts": header.get("ts", 0.0),
                "inference_ms": float(args.sleep_ms),
                "n_det": 0,
                "classes": [],
                "scores": [],
                "bboxes": [],
            })
            sock.send(reply)  # single-part reply
    except (KeyboardInterrupt, zmq.ContextTerminated):
        pass
    finally:
        sock.close(linger=0)
        ctx.term()
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
"""Handshake tests for SubprocessDetectorBridge (D-17).

Uses scripts/echo_detector_worker.py — the permanent repo helper from Plan 07 task 1.
Tests do NOT register any @detector_backend decorator — the bridge is skeleton-only
in Phase 2.
"""

import os
import signal
import sys
import time
import pathlib
import numpy as np
import pytest

REPO = pathlib.Path(__file__).resolve().parents[2]
ECHO_SCRIPT = str(REPO / "scripts" / "echo_detector_worker.py")


def _make_bridge(ipc_endpoint=None, sleep_ms=0):
    from src.perception.subprocess_bridge import SubprocessDetectorBridge
    # Invoke the echo worker via python <script> so chmod isn't required
    args = ["--sleep-ms", str(sleep_ms)] if sleep_ms else []
    # When the bridge Popens the subprocess, it passes ["--zmq", endpoint]
    # The echo worker reads those argv flags.
    # To use python as the binary + script as the first arg:
    #   bridge = SubprocessDetectorBridge(binary_path=sys.executable, args=[ECHO_SCRIPT, *args], ipc_endpoint=...)
    # Popen command becomes: [sys.executable, "--zmq", endpoint, ECHO_SCRIPT, ...extra_args]
    # That's WRONG — --zmq comes after the script name. Fix: bridge.start() does
    # cmd = [binary, "--zmq", endpoint] + args. So we need args = [script_path, "--sleep-ms", ...]
    # and binary = sys.executable. Then cmd = [python, "--zmq", endpoint, script, "--sleep-ms", N].
    # But python rejects --zmq before the script. Workaround: pass script as binary_path directly
    # if it's chmod+x'd, OR wrap in a small shell trick.
    #
    # Simpler: invoke via python -c is wrong per D-15. Correct: use chmod+x on the script,
    # then binary_path=ECHO_SCRIPT works (shebang handles python invocation).
    #
    # BUT Plan 07 Task 1's chmod is best-effort. Robust fallback: use a tiny wrapper:
    #   binary_path = sys.executable  + args = [ECHO_SCRIPT, ...sleep_args]
    # This means Popen cmd = [python, "--zmq", endpoint, ECHO_SCRIPT, --sleep-ms, N] which is WRONG.
    #
    # Correct fix: CustomBridge subclass that builds cmd as [python, script, --zmq, endpoint, ...args].
    # OR: call os.chmod in the test fixture to force executability, then binary_path=ECHO_SCRIPT.
    # OR (cleanest): modify SubprocessDetectorBridge to accept a cmd_builder callable (overkill for Phase 2).
    #
    # Use the os.chmod approach — belt-and-suspenders with Plan 07 Task 1's chmod:
    os.chmod(ECHO_SCRIPT, 0o755)
    extra_args = ["--sleep-ms", str(sleep_ms)] if sleep_ms else []
    bridge = SubprocessDetectorBridge(
        binary_path=ECHO_SCRIPT, args=extra_args, ipc_endpoint=ipc_endpoint,
    )
    return bridge


@pytest.fixture
def bridge():
    b = _make_bridge()
    yield b
    try:
        b.shutdown()
    except Exception:
        pass


def _dummy_frame():
    rgb = np.zeros((480, 640, 3), dtype=np.uint8)
    depth = np.full((480, 640), 2.0, dtype=np.float32)
    return rgb, depth


def test_endpoint_prefix_distinct_from_slam():
    from src.perception.subprocess_bridge import SubprocessDetectorBridge
    b = SubprocessDetectorBridge(binary_path="/bin/true")
    assert "detector_bridge_" in b.endpoint
    assert "slam_bridge_" not in b.endpoint  # D-16


def test_spawn_send_recv_round_trip(bridge):
    bridge.start()
    time.sleep(0.3)  # give echo worker time to connect
    rgb, depth = _dummy_frame()
    reply = bridge.send_frame(rgb, depth, timestamp=1.0)
    assert reply is not None
    # Keys may be str (raw=False) — assert str keys only
    assert set(reply.keys()) >= {"ts", "inference_ms", "n_det", "classes", "scores", "bboxes"}
    assert reply["n_det"] == 0
    assert reply["classes"] == []
    assert reply["scores"] == []
    assert reply["bboxes"] == []


def test_msgpack_header_fidelity(bridge):
    bridge.start()
    time.sleep(0.3)
    rgb, depth = _dummy_frame()
    reply = bridge.send_frame(rgb, depth, timestamp=7.5, params={"confidence": 0.7})
    assert reply is not None
    assert abs(reply["ts"] - 7.5) < 1e-9


def test_slow_backend_still_returns_before_timeout():
    b = _make_bridge(sleep_ms=200)
    try:
        b.start()
        time.sleep(0.3)
        rgb, depth = _dummy_frame()
        t0 = time.monotonic()
        reply = b.send_frame(rgb, depth, timestamp=2.0)
        elapsed = time.monotonic() - t0
        assert reply is not None
        assert elapsed < 1.0  # 200 ms + overhead, well under 5 s
    finally:
        b.shutdown()


def test_kill_triggers_zmq_again_within_timeout(bridge):
    bridge.start()
    time.sleep(0.3)
    rgb, depth = _dummy_frame()
    # First send succeeds
    assert bridge.send_frame(rgb, depth, timestamp=1.0) is not None
    # Externally kill the echo worker
    if bridge._process is not None:
        os.kill(bridge._process.pid, signal.SIGKILL)
        time.sleep(0.2)  # let the kernel deliver the signal
    # Next send: worker gone, recv times out within HANG_TIMEOUT_MS (+ slack)
    t0 = time.monotonic()
    reply = bridge.send_frame(rgb, depth, timestamp=2.0)
    elapsed = time.monotonic() - t0
    assert reply is None
    # Timeout path: either Popen.poll path exits fast, OR zmq.Again fires in ~5 s
    assert elapsed < bridge.HANG_TIMEOUT_MS / 1000.0 + 1.0
    assert bridge.alive is False


def test_shutdown_cleans_up_ipc_socket_file():
    endpoint = f"ipc:///tmp/detector_bridge_test_{os.getpid()}_{time.time_ns()}"
    sock_path = endpoint.replace("ipc://", "")
    b = _make_bridge(ipc_endpoint=endpoint)
    try:
        b.start()
        time.sleep(0.3)
        assert os.path.exists(sock_path), "IPC socket file should exist while bridge is alive"
    finally:
        b.shutdown()
    assert not os.path.exists(sock_path), (
        f"IPC socket file leaked at {sock_path} after shutdown (Pitfall 3)"
    )
```
