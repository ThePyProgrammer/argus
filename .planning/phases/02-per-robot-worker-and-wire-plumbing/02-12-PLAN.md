```python
"""Pool end-to-end parity test — replaces test_yolov11_regression.py (D-19).

Proves DetectorWorkerPool.submit(rid, frame, pose) → latest(rid) produces
the SAME detections as calling YOLOv11Backend.process_frame + MedianDepthLifter
directly. This is the Phase 2 pool-level regression guard.
"""

import time
import numpy as np
import pytest
from pathlib import Path

ultralytics = pytest.importorskip("ultralytics")
_ = pytest.importorskip("torch")

REPO = Path(__file__).resolve().parents[2]
FIXTURE_PATH = REPO / "tests" / "fixtures" / "yolo_regression_scene_01.npz"


@pytest.fixture(autouse=True)
def _clean_registries():
    import sys
    from src.perception.registry import DetectorRegistry, Detection3DRegistry
    DetectorRegistry._clear(); Detection3DRegistry._clear()
    yield
    DetectorRegistry._clear(); Detection3DRegistry._clear()
    for mod in ("src.perception.backends", "src.perception.lifters",
                "src.perception.backends.yolov11_backend",
                "src.perception.lifters.median_depth"):
        sys.modules.pop(mod, None)


@pytest.fixture
def fixture_frame():
    if not FIXTURE_PATH.exists():
        pytest.skip(f"Fixture not generated: {FIXTURE_PATH}")
    data = np.load(FIXTURE_PATH)
    return {
        "rgb": data["rgb"],
        "depth": data["depth"],
        "pose": data["pose"],
    }


def test_pool_submit_latest_matches_direct_backend(fixture_frame):
    import src.perception.backends  # noqa: F401 — trigger @detector_backend
    import src.perception.lifters   # noqa: F401 — trigger @detection_3d
    from src.perception.worker_pool import DetectorWorkerPool
    from src.perception.registry import DetectorRegistry, Detection3DRegistry
    from src.bridge.sensor_types import SensorFrame, CameraIntrinsics

    intrinsics = CameraIntrinsics.from_fov(640, 480, 70.0)
    rgb = fixture_frame["rgb"]
    depth = fixture_frame["depth"]
    pose = fixture_frame["pose"]

    # Direct backend + lifter reference
    direct_det = DetectorRegistry.create("yolov11")
    direct_lift = Detection3DRegistry.create("median_depth")
    frame = SensorFrame(rgb=rgb, depth=depth, ground_truth_pose=np.eye(4), sim_time=0.1)
    direct_2d = direct_det.process_frame(frame)
    direct_3d = direct_lift.lift(direct_2d, frame, pose, intrinsics, None)

    # Skip if fixture has zero detections (W-01 parity would be meaningless)
    if len(direct_3d.items) == 0:
        pytest.skip("Fixture produces 0 YOLO detections — W-01 fix required")

    # Pool dataflow
    pool = DetectorWorkerPool(
        robot_ids=["r0"],
        backend_name="yolov11",
        backend_params={},
        lifter_name="median_depth",
        intrinsics_per_robot={"r0": intrinsics},
    )
    pool.start()
    try:
        frame2 = SensorFrame(rgb=rgb, depth=depth, ground_truth_pose=np.eye(4), sim_time=0.2)
        pool.submit("r0", frame2, pose, None)
        # Poll for latest within 5 s
        deadline = time.monotonic() + 5.0
        pool_3d = None
        while time.monotonic() < deadline:
            pool_3d = pool.latest("r0")
            if pool_3d is not None:
                break
            time.sleep(0.05)
        assert pool_3d is not None, "pool.latest timed out after 5 s"
    finally:
        pool.shutdown()

    # Parity
    assert len(pool_3d.items) == len(direct_3d.items), (
        f"detection count diverged: pool={len(pool_3d.items)} direct={len(direct_3d.items)}"
    )
    for i, (p_item, d_item) in enumerate(zip(pool_3d.items, direct_3d.items)):
        assert p_item.class_id == d_item.class_id, f"item {i}: class_id differs"
        np.testing.assert_allclose(p_item.center, d_item.center, atol=1e-6,
                                   err_msg=f"item {i}: center differs")
        np.testing.assert_allclose(p_item.half_extents, d_item.half_extents, atol=1e-6,
                                   err_msg=f"item {i}: half_extents differs")
```
