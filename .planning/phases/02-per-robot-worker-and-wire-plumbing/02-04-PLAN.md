Class body:

```python
class DetectorWorkerPool:
    def __init__(
        self,
        robot_ids: list[str],
        backend_name: str,
        backend_params: dict | None,
        lifter_name: str,
        intrinsics_per_robot: dict[str, "CameraIntrinsics"],
    ) -> None:
        self.backend_name = backend_name
        self.lifter_name = lifter_name
        self._workers: dict[str, DetectorWorker] = {}
        params = dict(backend_params or {})
        for rid in robot_ids:
            detector = DetectorRegistry.create(backend_name, **params)
            lifter = Detection3DRegistry.create(lifter_name)
            self._workers[rid] = DetectorWorker(rid, detector, lifter, intrinsics_per_robot[rid])

    def start(self) -> None:
        for w in self._workers.values():
            w.start()

    def warmup_all(self, dummy_frames: dict[str, "SensorFrame"]) -> None:
        """D-03: synchronous warmup — main.py restart block awaits this before
        emitting detector_restart_complete so the UI's restart overlay dismisses
        only after first inference is real (no oneDNN cold-start stall on next real frame).
        """
        for rid, w in self._workers.items():
            frame = dummy_frames.get(rid)
            if frame is None:
                logger.warning("warmup_all: no dummy frame for robot %s; skipping", rid)
                continue
            try:
                w.warmup(frame)
            except Exception as exc:
                logger.warning("warmup_all failed for %s: %s", rid, exc)

    def submit(self, rid: str, frame, pose: np.ndarray, slam_cloud: np.ndarray | None = None) -> None:
        w = self._workers.get(rid)
        if w is None:
            return  # D-05: unknown robot = safe no-op
        w.submit(frame, pose, slam_cloud)

    def latest(self, rid: str) -> "Detections3D | None":
        w = self._workers.get(rid)
        return w.latest() if w is not None else None

    def inspect_worker_queues(self) -> dict[str, dict]:
        return {rid: w.inspect() for rid, w in self._workers.items()}

    def reset_all(self) -> None:
        for w in self._workers.values():
            w.reset()

    def shutdown(self) -> None:
        for w in self._workers.values():
            w.shutdown()

    @property
    def robot_ids(self) -> list[str]:
        return list(self._workers.keys())
```

Rationale for NOT calling start() inside __init__: main.py constructs the pool, then calls warmup_all, then start() — but in practice the worker threads may need to be running to service warmup calls if we redesigned later. For Phase 2, warmup is a DIRECT call on the detector (bypassing the queue); start() is separate. Document this in the pool docstring so Wave 4 wires `pool = DetectorWorkerPool(...); pool.warmup_all({...}); pool.start()`.
