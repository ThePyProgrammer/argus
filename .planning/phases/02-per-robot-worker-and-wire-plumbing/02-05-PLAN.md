```python
import time
import numpy as np
import pytest

from src.perception.worker import DetectorWorker
from src.perception.types import Detections2D, Detections3D, OrientedBox3D
from src.bridge.sensor_types import SensorFrame, CameraIntrinsics


class SlowDetector:
    CAPABILITIES = {}
    PARAMETER_SCHEMA = {}
    def warmup(self, f): pass
    def reset(self): pass
    def process_frame(self, f):
        time.sleep(0.3)
        return Detections2D(items=[], inference_ms=300.0, image_hw=(480, 640))
    def get_metrics(self): return {}
    def apply_params(self, p): return {}


class EmptyLifter:
    """Returns Detections3D with defaulted capture_pose/capture_timestamp; worker overwrites them."""
    def lift(self, d2d, f, pose, intr, cloud):
        return Detections3D(
            items=[], lifter_ms=0.0, detector_ms=d2d.inference_ms,
            n_raw=0, n_final=0, image_hw=d2d.image_hw,
            capture_pose=np.eye(4),          # lifter's stale default
            capture_timestamp=0.0,            # lifter's stale default
        )
    def reset(self): pass


class ItemsLifter:
    """Returns Detections3D with 3 OBB items so test can verify envelope-level pose."""
    def lift(self, d2d, f, pose, intr, cloud):
        items = [
            OrientedBox3D(
                center=np.array([i, 0, 0], dtype=np.float64),
                half_extents=np.array([0.5, 0.5, 0.5], dtype=np.float64),
                quaternion=np.array([0, 0, 0, 1], dtype=np.float64),
                class_id=56, class_name="chair", score=0.9, track_id=None,
            )
            for i in range(3)
        ]
        return Detections3D(
            items=items, lifter_ms=1.0, detector_ms=d2d.inference_ms,
            n_raw=3, n_final=3, image_hw=d2d.image_hw,
            capture_pose=np.eye(4), capture_timestamp=0.0,
        )
    def reset(self): pass


def _make_frame(sim_time: float) -> SensorFrame:
    rgb = np.zeros((480, 640, 3), dtype=np.uint8)
    depth = np.full((480, 640), 2.0, dtype=np.float32)
    return SensorFrame(rgb=rgb, depth=depth, ground_truth_pose=np.eye(4), sim_time=sim_time)


def test_capture_pose_snapshotted_at_submit():
    intr = CameraIntrinsics.from_fov(640, 480, 70.0)
    w = DetectorWorker("r0", SlowDetector(), EmptyLifter(), intr)
    w.start()
    try:
        # Pose at submit time: translation 5
        pose_at_submit = np.eye(4)
        pose_at_submit[0, 3] = 5.0
        frame = _make_frame(sim_time=1.0)
        w.submit(frame, pose_at_submit, None)
        # Before worker's _loop finishes inference, simulate the caller advancing the pose
        # to translation 99 in place (mimicking robot motion during 300 ms inference)
        pose_at_submit[0, 3] = 99.0
        time.sleep(0.5)  # wait for inference to complete
        latest = w.latest()
        assert latest is not None, "worker did not produce a Detections3D"
        assert latest.capture_pose[0, 3] == 5.0, (
            f"capture_pose drifted: expected 5.0 (submit-time), got {latest.capture_pose[0,3]} (lift-time). "
            f"D-13 contract violated."
        )
    finally:
        w.shutdown()


def test_capture_timestamp_from_frame_sim_time():
    intr = CameraIntrinsics.from_fov(640, 480, 70.0)
    w = DetectorWorker("r0", SlowDetector(), EmptyLifter(), intr)
    w.start()
    try:
        frame = _make_frame(sim_time=42.5)
        w.submit(frame, np.eye(4), None)
        time.sleep(0.5)
        latest = w.latest()
        assert latest is not None
        assert latest.capture_timestamp == 42.5, (
            f"capture_timestamp drift: expected 42.5, got {latest.capture_timestamp}. "
            f"D-12 contract violated."
        )
    finally:
        w.shutdown()


def test_envelope_level_not_per_box():
    intr = CameraIntrinsics.from_fov(640, 480, 70.0)
    w = DetectorWorker("r0", SlowDetector(), ItemsLifter(), intr)
    w.start()
    try:
        pose = np.eye(4)
        pose[0, 3] = 7.0
        frame = _make_frame(sim_time=3.14)
        w.submit(frame, pose, None)
        time.sleep(0.5)
        latest = w.latest()
        assert latest is not None
        # Envelope-level: one capture_pose shared by all 3 items
        assert len(latest.items) == 3
        assert latest.capture_pose[0, 3] == 7.0
        assert latest.capture_timestamp == 3.14
        # OrientedBox3D has NO per-box capture_pose field — D-11 invariant
        for i, item in enumerate(latest.items):
            assert not hasattr(item, "capture_pose"), (
                f"box {i}: OrientedBox3D must not carry per-box capture_pose (D-11)"
            )
    finally:
        w.shutdown()
```
