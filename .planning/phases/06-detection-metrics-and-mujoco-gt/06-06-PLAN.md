    ## Step 1 — `src/metrics/detection_export.py`

    ```python
    """Append-on-emit JSONL writer for detection export (DET-METRICS-04).

    Per CONTEXT D-11 / D-12:
    - Writes to <tempdir>/argus_sessions/<session_id>/detections.jsonl in append
      + line-buffered text mode so SIGKILL preserves flushed lines.
    - Session ID is UUID4 generated by the caller (coordinator boot).
    - rotate() closes the current file and opens a new session directory;
      prior files are NOT deleted (debug artifacts per D-12).
    - Thread-safe: append and rotate share a threading.Lock
      (RESEARCH Open Question 4 — rotate can fire from a REST handler thread
      while append runs on the coordinator thread).

    Security (T-6-02): session path uses server-generated UUID4 via
    tempfile.gettempdir() — no user input on the filesystem path. Callers MUST
    NOT pass user-supplied strings as session_id.
    """
    from __future__ import annotations
    import json
    import tempfile
    import threading
    from pathlib import Path
    from typing import Any


    class DetectionExportWriter:
        def __init__(self, session_id: str, base_dir: Path | None = None) -> None:
            self._session_id = str(session_id)
            self._base_dir = Path(base_dir) if base_dir is not None else Path(tempfile.gettempdir())
            self._lock = threading.Lock()
            self._fp = None
            self._file_path: Path | None = None
            self._open_for_session(self._session_id)

        def _open_for_session(self, session_id: str) -> None:
            session_dir = self._base_dir / "argus_sessions" / session_id
            session_dir.mkdir(parents=True, exist_ok=True)
            path = session_dir / "detections.jsonl"
            # Text + line-buffered (buffering=1) so "\n" writes flush (Pitfall 6).
            # If file doesn't exist, this creates it empty.
            self._fp = open(path, "a", buffering=1, encoding="utf-8")
            self._file_path = path
            self._session_id = session_id

        @property
        def session_id(self) -> str:
            return self._session_id

        @property
        def file_path(self) -> Path:
            assert self._file_path is not None
            return self._file_path

        def append(
            self,
            obb,  # OrientedBox3D
            robot_id: str,
            backend_id: str,
            capture_timestamp: float,
        ) -> None:
            record = {
                "robot_id": str(robot_id),
                "backend_id": str(backend_id),
                "capture_timestamp": float(capture_timestamp),
                "obb": obb.to_wire(),  # Phase 1 D-10: sole quaternion path
            }
            line = json.dumps(record, separators=(",", ":")) + "\n"
            with self._lock:
                if self._fp is None:
                    # Writer has been closed; silently no-op rather than crash.
                    return
                self._fp.write(line)

        def rotate(self, new_session_id: str) -> None:
            with self._lock:
                if self._fp is not None:
                    self._fp.close()
                    self._fp = None
                self._open_for_session(str(new_session_id))

        def close(self) -> None:
            with self._lock:
                if self._fp is not None:
                    self._fp.close()
                    self._fp = None
    ```
