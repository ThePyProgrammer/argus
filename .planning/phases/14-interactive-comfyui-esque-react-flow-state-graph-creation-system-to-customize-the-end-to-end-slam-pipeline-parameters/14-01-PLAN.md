   import type { Node, Edge, OnNodesChange, OnEdgesChange, OnConnect, XYPosition } from '@xyflow/react';

   // Port data types (per CONTEXT.md locked decision: typed ports with visual hints)
   export type PortDataType = 'Image' | 'PointCloud' | 'Pose' | 'IMU' | 'Scalar' | 'Boolean' | 'Config';

   // Port definition for node inputs/outputs
   export interface PortDef {
     id: string;
     label: string;
     dataType: PortDataType;
     required: boolean;  // unconnected required ports = validation error
   }

   // Node categories (per CONTEXT.md: 4 categories, expanded to 7 for granularity)
   export type NodeCategory = 'sensor' | 'slam' | 'merger' | 'filter' | 'splitter' | 'parameter' | 'output';

   // Custom node data shape
   export interface PipelineNodeData {
     label: string;
     category: NodeCategory;
     headerColor: string;
     inputs: PortDef[];
     outputs: PortDef[];
     parameterSchema: Record<string, unknown> | null;  // JSON Schema properties subset
     paramValues: Record<string, unknown>;  // current param values
     status: 'idle' | 'processing' | 'error' | 'initializing';
     registryName?: string;  // e.g. 'icp', 'orbslam3' -- links to SLAMRegistry/MergeRegistry
     nodeType: string;  // definition key, e.g. 'sensor_rgbd', 'slam_icp'
   }

   // React Flow node with our custom data
   export type PipelineNode = Node<PipelineNodeData>;

   // Custom edge data shape
   export interface PipelineEdgeData {
     fps: number;       // throughput (0 = idle)
     dataType: PortDataType;
   }

   export type PipelineEdge = Edge<PipelineEdgeData>;

   // Preset metadata
   export interface PresetInfo {
     name: string;
     builtIn: boolean;
     nodeCount: number;
   }

   // Validation error
   export interface ValidationError {
     nodeId?: string;
     message: string;
   }

   // Serialized graph config for backend
   export interface PipelineConfig {
     nodes: Array<{ id: string; type: string; params: Record<string, unknown>; position: { x: number; y: number } }>;
     edges: Array<{ source: string; sourceHandle: string; target: string; targetHandle: string }>;
   }

   // Node definition (template for creating instances)
   export interface NodeDefinition {
     type: string;          // unique key, e.g. 'sensor_rgbd'
     label: string;
     category: NodeCategory;
     inputs: PortDef[];
     outputs: PortDef[];
     defaultParams: Record<string, unknown>;
     parameterSchema: Record<string, unknown> | null;
   }
   ```

1. Create `frontend/src/utils/pipelineValidation.ts`:

   - `detectCycles(nodes: PipelineNode[], edges: PipelineEdge[]): string[] | null` -- Kahn's algorithm (topological sort). Returns array of node labels in cycle, or null if DAG is valid. Implementation:
     ```
     Build inDegree map and adjacency list from edges.
     Initialize queue with nodes having inDegree 0.
     Process queue: for each node, decrement neighbors' inDegree; add to queue if 0.
     If sorted.length !== nodes.length, return labels of nodes NOT in sorted.
     ```
   - `findUnconnectedPorts(nodes: PipelineNode[], edges: PipelineEdge[]): ValidationError[]` -- For each node, check each input port with `required: true`. If no edge targets that port, return error with message `"Node '{label}' has unconnected required input '{port.label}'"`.
   - `findMissingOutput(nodes: PipelineNode[]): ValidationError | null` -- Check if at least one node has category 'output'. If not, return error `"Pipeline has no visualization output node"`.
   - `validateGraph(nodes: PipelineNode[], edges: PipelineEdge[]): ValidationError[]` -- Runs all three checks, returns combined array. Order: cycles first, then unconnected, then missing output.

2. Create `frontend/src/utils/pipelineSerializer.ts`:

   - `serializeGraph(nodes: PipelineNode[], edges: PipelineEdge[]): PipelineConfig` -- Maps nodes to `{id, type: node.data.nodeType, params: resolvedParams, position: node.position}`. For each node, resolvedParams starts from `node.data.paramValues`. Then traverse edges: if an edge source node has category 'parameter', the parameter node's `paramValues.value` overrides the target node's param matching the target handle name. Maps edges to `{source, sourceHandle, target, targetHandle}`.
   - `deserializeGraph(config: PipelineConfig, existingDefinitions: Record<string, NodeDefinition>): { nodes: PipelineNode[], edges: PipelineEdge[] }` -- Reconstructs React Flow nodes from config using `buildNodeData()`. Assigns positions from config. Reconstructs edges with `type: 'animated'` and default edgeData `{fps: 0, dataType: 'PointCloud'}`.

   nodes: PipelineNode[]              -- default []
   edges: PipelineEdge[]              -- default []
   selectedNodeId: string | null      -- default null
   activePreset: string | null        -- default null
   availablePresets: PresetInfo[]      -- default []
   isDirty: boolean                   -- default false
   validationErrors: ValidationError[] -- default []
   isValid: boolean                   -- default true
   isApplying: boolean                -- default false

   ```

   Actions:
   - `onNodesChange: OnNodesChange` -- calls `applyNodeChanges(changes, state.nodes)`, sets isDirty true
   - `onEdgesChange: OnEdgesChange` -- calls `applyEdgeChanges(changes, state.edges)`, sets isDirty true
   - `onConnect: OnConnect` -- calls `addEdge({...connection, type:'animated', data:{fps:0, dataType:'PointCloud'}}, state.edges)`, sets isDirty true
   - `selectNode(id: string | null)` -- sets selectedNodeId
   - `addNode(defKey: string, position: XYPosition, registryName?: string, registrySchema?: Record<string, unknown>)` -- creates new PipelineNode with unique id (e.g. `${defKey}_${Date.now()}`), data from `buildNodeData(defKey, registryName, registrySchema)`, position. Appends to nodes. Sets isDirty true.
   - `removeNode(id: string)` -- filters out node and its connected edges. Sets isDirty true. Clears selectedNodeId if it was the removed node.
   - `updateNodeParam(nodeId: string, key: string, value: unknown)` -- updates `node.data.paramValues[key]` for the matching node. Sets isDirty true.

   - `setEdgeThroughput(edgeId: string, fps: number)` -- updates edgeThroughputs record

   - `loadPresetGraph(nodes: PipelineNode[], edges: PipelineEdge[], presetName: string)` -- replaces nodes/edges, sets activePreset, isDirty false
   - `resetToLastApplied()` -- if lastAppliedConfig exists, deserialize and replace nodes/edges, set isDirty false
