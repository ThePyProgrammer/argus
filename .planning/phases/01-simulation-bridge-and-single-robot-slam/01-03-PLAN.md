       ```python
       import numpy as np
       import open3d as o3d
       from src.bridge.sensor_types import CameraIntrinsics

       def depth_to_pointcloud(
           depth: np.ndarray,
           rgb: np.ndarray,
           intrinsics: CameraIntrinsics,
           max_depth: float = 10.0,
       ) -> o3d.geometry.PointCloud:
           h, w = depth.shape
           u, v = np.meshgrid(np.arange(w), np.arange(h))
           valid = (depth > 0) & (depth < max_depth)
           z = depth[valid]
           x = (u[valid] - intrinsics.cx) * z / intrinsics.fx
           y = (v[valid] - intrinsics.cy) * z / intrinsics.fy
           points = np.stack([x, y, z], axis=-1)
           colors = rgb[valid].astype(np.float64) / 255.0
           pcd = o3d.geometry.PointCloud()
           pcd.points = o3d.utility.Vector3dVector(points)
           pcd.colors = o3d.utility.Vector3dVector(colors)
           return pcd
       ```

       ```python
       import octomap
       import numpy as np

       class OctoMapBuilder:
           def __init__(self, resolution: float = 0.1):
               self._tree = octomap.OcTree(resolution)
               self._resolution = resolution

           def insert_scan(self, points: np.ndarray, sensor_origin: np.ndarray):
               # points: (N, 3) float64 in global frame
               # sensor_origin: (3,) float64
               self._tree.insertPointCloud(
                   pointcloud=points.astype(np.float64),
                   origin=sensor_origin.astype(np.float64),
               )

           def get_occupied_voxels(self) -> np.ndarray:
               occupied = []
               for node in self._tree.begin_tree():
                   if self._tree.isNodeOccupied(node):
                       occupied.append([node.getX(), node.getY(), node.getZ()])
               return np.array(occupied) if occupied else np.empty((0, 3))

           @property
           def resolution(self) -> float:
               return self._resolution

           @property
           def num_occupied(self) -> int:
               return len(self.get_occupied_voxels())
       ```

       ```python
       import numpy as np
       from evo.core.trajectory import PoseTrajectory3D
       from evo.core import metrics, sync

       def compute_drift_metrics(
           slam_poses: list[np.ndarray],
           gt_poses: list[np.ndarray],
           timestamps: list[float],
       ) -> dict:
           slam_traj = PoseTrajectory3D(
               poses_se3=slam_poses,
               timestamps=np.array(timestamps),
           )
           gt_traj = PoseTrajectory3D(
               poses_se3=gt_poses,
               timestamps=np.array(timestamps),
           )
           gt_synced, slam_synced = sync.associate_trajectories(gt_traj, slam_traj)
           ate_metric = metrics.APE(metrics.PoseRelation.translation_part)
           ate_metric.process_data((gt_synced, slam_synced))
           rpe_metric = metrics.RPE(metrics.PoseRelation.translation_part)
           rpe_metric.process_data((gt_synced, slam_synced))
           return {
               "ate_rmse": ate_metric.get_statistic(metrics.StatisticsType.rmse),
               "ate_mean": ate_metric.get_statistic(metrics.StatisticsType.mean),
               "rpe_rmse": rpe_metric.get_statistic(metrics.StatisticsType.rmse),
               "rpe_mean": rpe_metric.get_statistic(metrics.StatisticsType.mean),
           }
       ```

This lite pipeline is sufficient for Phase 1 (single robot) and can be replaced with RTAB-Map ROS 2 in later phases.

    ```python
    import numpy as np
    import open3d as o3d
    from src.bridge.sensor_types import SensorFrame, CameraIntrinsics
    from src.slam.depth_to_cloud import depth_to_pointcloud

    class SLAMPipeline:
        def __init__(self, intrinsics: CameraIntrinsics, voxel_size: float = 0.05):
            self._intrinsics = intrinsics
            self._voxel_size = voxel_size
            self._global_cloud = o3d.geometry.PointCloud()
            self._slam_poses: list[np.ndarray] = []
            self._prev_cloud: o3d.geometry.PointCloud | None = None
            self._current_pose: np.ndarray = np.eye(4)  # cumulative pose

        def process_frame(self, frame: SensorFrame) -> np.ndarray:
            """Process one RGB-D frame. Returns estimated (4,4) pose."""
            if frame.depth is None:
                # Monocular fallback: use ground-truth pose, no SLAM
                self._slam_poses.append(frame.ground_truth_pose.copy())
                return frame.ground_truth_pose

            # Convert depth to point cloud
            cloud = depth_to_pointcloud(
                frame.depth, frame.rgb, self._intrinsics, max_depth=10.0
            )
            cloud = cloud.voxel_down_sample(self._voxel_size)

            if self._prev_cloud is not None and len(cloud.points) > 100:
                # ICP odometry: align current to previous
                result = o3d.pipelines.registration.registration_icp(
                    cloud, self._prev_cloud,
                    max_correspondence_distance=self._voxel_size * 3,
                    init=np.eye(4),
                    estimation_method=o3d.pipelines.registration.TransformationEstimationPointToPoint(),
                )
                if result.fitness > 0.3:  # reasonable alignment
                    self._current_pose = self._current_pose @ result.transformation

            # Transform cloud to global frame and accumulate
            cloud.transform(self._current_pose)
            self._global_cloud += cloud
            # Downsample accumulated cloud periodically (every 10 frames)
            if len(self._slam_poses) % 10 == 0:
                self._global_cloud = self._global_cloud.voxel_down_sample(self._voxel_size)

            self._slam_poses.append(self._current_pose.copy())
            self._prev_cloud = depth_to_pointcloud(
                frame.depth, frame.rgb, self._intrinsics, max_depth=10.0
            ).voxel_down_sample(self._voxel_size)

            return self._current_pose.copy()

        @property
        def global_cloud(self) -> o3d.geometry.PointCloud:
            return self._global_cloud

        @property
        def slam_poses(self) -> list[np.ndarray]:
            return list(self._slam_poses)

        @property
        def num_frames_processed(self) -> int:
            return len(self._slam_poses)

        def get_cloud_points(self) -> np.ndarray:
            """Get global cloud as (N, 3) numpy array."""
            return np.asarray(self._global_cloud.points)

        def get_cloud_colors(self) -> np.ndarray:
            """Get global cloud colors as (N, 3) numpy array."""
            return np.asarray(self._global_cloud.colors)
    ```
