    """DET-MODELS-06 — End-to-end kill -9 → fallback within 5 s (Plan 05-12 activation).

    @pytest.mark.slow_boxer — requires subprocess_venvs/boxer/.ready + models/boxer/<SHA>/.
    Skipped by default; run explicitly via `pytest -m slow_boxer`.

    Test flow:
      1. Pre-check: subprocess_venvs/boxer/.ready exists (setup script ran).
      2. Instantiate BoxeRBackend → warmup(dummy_frame) → bridge spawns worker.
      3. Grab the worker pid via backend._bridge._process.pid.
      4. os.kill(pid, signal.SIGKILL) — simulate hard crash.
      5. Call backend.process_frame(frame) → expect BridgeHangError or SubprocessDiedError.
      6. Assert elapsed time from kill to exception < 5.5 s (SC#3 budget + buffer).
      7. Construct a DetectorWorkerPool with the backend and a fake streaming_viz;
         register pool as worker._pool_ref; call worker._loop's exception path
         manually OR call pool.on_backend_crash directly; assert crash_fallback
         message on streaming_viz._message_queue AND pool.backend_name == "yolov11".
    """
    from __future__ import annotations

    import os
    import signal
    import time
    from pathlib import Path

    import numpy as np
    import pytest

    pytestmark = pytest.mark.slow_boxer


    @pytest.mark.slow_boxer
    def test_kill_nine_triggers_fallback_within_5s() -> None:
        """SC#3 — kill -9 BoxeR subprocess → fallback completes within 5 s."""
        from src.bridge.sensor_types import CameraIntrinsics, SensorFrame
        import src.perception.backends  # noqa: F401
        from src.perception.backends.boxer_backend import BoxeRBackend
        from src.perception.subprocess_bridge import BridgeHangError, SubprocessDiedError
        from src.perception.worker_pool import DetectorWorkerPool

        # Precondition: subprocess venv must be ready. Skip if not provisioned.
        if not Path("subprocess_venvs/boxer/.ready").exists():
            pytest.skip("subprocess_venvs/boxer/.ready missing — run `make download-models-boxer` first.")

        dummy = SensorFrame(
            rgb=np.zeros((480, 640, 3), dtype=np.uint8),
            depth=np.zeros((480, 640), dtype=np.float32),
            ground_truth_pose=np.eye(4),
            sim_time=0.0,
        )

        backend = BoxeRBackend()
        # warmup() lazy-spawns the bridge and runs one inference
        backend.warmup(dummy)
        assert backend._bridge is not None
        assert backend._bridge._process is not None
        worker_pid = backend._bridge._process.pid
        assert worker_pid is not None and worker_pid > 0

        # Hard kill the worker — simulates SC#3 scenario.
        t_kill = time.perf_counter()
        os.kill(worker_pid, signal.SIGKILL)

        # The next send_frame must raise BridgeHangError or SubprocessDiedError
        # within HANG_TIMEOUT_MS (5 s) + handshake buffer.
        with pytest.raises((BridgeHangError, SubprocessDiedError)):
            backend.process_frame(dummy)
        elapsed = time.perf_counter() - t_kill
        assert elapsed < 5.5, f"elapsed {elapsed:.2f} s exceeds SC#3 budget 5.0 s (+0.5 s slack)"

        # Simulate the pool handler path: construct a minimal pool, wire streaming_viz,
        # and call on_backend_crash — verify fallback completes end-to-end.
        # NOTE: This uses a test-local FakeStreamingViz stub, NOT the real
        # streaming_viz wired in main.py (Plan 10). That is intentional — Plan 12
        # exercises pool.on_backend_crash (Plan 09) in isolation, with no
        # dependency on main.py or the frontend. See plan 05-12 depends_on=[09,11].
        class FakeStreamingViz:
            def __init__(self):
                self._message_queue: list[dict] = []
        fake_viz = FakeStreamingViz()
        intr = CameraIntrinsics(fx=500.0, fy=500.0, cx=320.0, cy=240.0, width=640, height=480)
        pool = DetectorWorkerPool(
            robot_ids=["r1"],
            backend_name="yolov11",
            backend_params={},
            lifter_name="median_depth",
            intrinsics_per_robot={"r1": intr},
            streaming_viz=fake_viz,
        )
        pool.on_backend_crash(crashed_backend="boxer", reason="SIGKILL test")
        assert pool.backend_name == "yolov11"
        assert len(fake_viz._message_queue) == 1
        assert fake_viz._message_queue[0]["type"] == "crash_fallback"
        assert fake_viz._message_queue[0]["payload"]["subsystem"] == "detector"

        # Cleanup.
        backend.shutdown()
    ```

    """DET-MODELS-07 — Offline boot gate (Plan 05-12 activation).

    @pytest.mark.slow_boxer — requires full model tree + subprocess_venvs/boxer/.
    Skipped by default; run explicitly via `pytest -m slow_boxer`.

    Test approach: subprocess-spawn a short Python snippet under
    HF_HUB_OFFLINE=1 + HF_HUB_DISABLE_TELEMETRY=1 + HF_HOME=/tmp/empty-hf-cache
    that imports src.perception.backends and enumerates DetectorRegistry.
    Assert YOLOv11, RTDETRv2, and BoxeR all report available=True.

    This catches regressions like:
      * rtdetrv2_backend.py silently reaches out to the HF hub at import time.
      * boxer_backend.py opens a network socket during available() probe.
      * onnxruntime tries to download providers at import time.
    """
    from __future__ import annotations

    import json
    import os
    import subprocess
    import sys
    import tempfile
    from pathlib import Path

    import pytest

    REPO_ROOT = Path(__file__).resolve().parent.parent.parent


    @pytest.mark.slow_boxer
    def test_all_backends_boot_with_hf_hub_offline() -> None:
        """SC#4 — all 3 backends enumerable + available with HF_HUB_OFFLINE=1."""
        # Preconditions: all three backends' artifacts must be pre-fetched.
        onnx_path = REPO_ROOT / "models" / "rtdetrv2"
        boxer_ready = REPO_ROOT / "subprocess_venvs" / "boxer" / ".ready"
        yolo_cache = Path.home() / ".cache" / "ultralytics"  # best-effort — YOLO caches weights here

        if not onnx_path.exists():
            pytest.skip(f"{onnx_path} missing — run `make download-models-rtdetrv2` first.")
        if not boxer_ready.exists():
            pytest.skip(f"{boxer_ready} missing — run `make download-models-boxer` first.")

        with tempfile.TemporaryDirectory() as empty_hf_cache:
            env = os.environ.copy()
            env["HF_HUB_OFFLINE"] = "1"
            env["HF_HUB_DISABLE_TELEMETRY"] = "1"
            env["HF_HOME"] = str(empty_hf_cache)
            env["TRANSFORMERS_OFFLINE"] = "1"

            probe_script = r"""
    import json
    import src.perception.backends  # noqa: F401 -- side-effect registration
    from src.perception.registry import DetectorRegistry
    entries = DetectorRegistry.list_backends()
    print(json.dumps({
        e["name"]: {"available": e["available"], "reason": e.get("reason")}
        for e in entries
    }))
    """
            result = subprocess.run(
                [sys.executable, "-c", probe_script],
                cwd=str(REPO_ROOT),
                env=env,
                capture_output=True, text=True, timeout=60,
            )
            assert result.returncode == 0, (
                f"probe failed. stdout={result.stdout!r} stderr={result.stderr!r}"
            )
            entries = json.loads(result.stdout.strip().splitlines()[-1])
            for name in ("yolov11", "rtdetrv2", "boxer"):
                assert name in entries, f"{name} missing from registry enumeration"
                assert entries[name]["available"], (
                    f"{name} not available under HF_HUB_OFFLINE=1: reason={entries[name]['reason']}"
                )
    ```
